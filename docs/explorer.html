def circle_positions(n: int) -> dict[int, tuple[float, float]]:
    angles = np.linspace(0, 2 * np.pi, max(n, 1), endpoint=False)
    return {i: (float(math.cos(angles[i])), float(math.sin(angles[i]))) for i in range(n)}


def build_B(signs: list[int], edges: list[tuple[int, int, float]]) -> np.ndarray:
    n = len(signs)
    B = np.diag([float(s) for s in signs])
    for u, v, w in edges:
        val = -1.0 if math.isinf(w) else -math.cos(math.pi / w)
        B[u, v] = val
        B[v, u] = val
    return B


def bipartition(n: int, edges: list[tuple[int, int, float]]):
    adj = [[] for _ in range(n)]
    for u, v, _ in edges:
        adj[u].append(v)
        adj[v].append(u)
    color = [-1] * n
    n1, n2 = [], []
    for start in range(n):
        if color[start] != -1:
            continue
        color[start] = 0
        q = deque([start])
        while q:
            u = q.popleft()
            (n1 if color[u] == 0 else n2).append(u)
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = 1 - color[u]
                    q.append(v)
                elif color[v] == color[u]:
                    return None
    return n1, n2


def principal_element(B: np.ndarray, order: list[int]) -> tuple[np.ndarray, np.ndarray]:
    P = B[np.ix_(order, order)]
    U_ord = np.triu(P)
    w_ord = -np.linalg.inv(U_ord) @ U_ord.T
    inv_perm = np.argsort(order)
    return U_ord, w_ord[np.ix_(inv_perm, inv_perm)]

def server(input, output, session):
    _, init_s, init_e = PRESETS["example_3_33"]
    init_signs, init_edges = list(init_s), list(init_e)
    qs = parse_qs(session.clientdata.url_search().lstrip("?")) if hasattr(session, "clientdata") else {}
    if "signs" in qs and qs["signs"]:
        s_parsed = [1 if ch == "+" else -1 for ch in qs["signs"][0] if ch in ("+", "-")]
        if s_parsed:
            init_signs = s_parsed
            raw_e = qs.get("edges", [""])[0]
            parsed_e = []
            for item in raw_e.split(","):
                item = item.strip()
                if not item or "-" not in item:
                    continue
                pair_s, w_s = (item.split(":", 1) + ["3"])[:2]
                u_s, v_s = pair_s.split("-", 1)
                u, v = int(u_s), int(v_s)
                w = math.inf if w_s.strip().lower() in ("oo", "inf") else float(w_s)
                if 0 <= u < len(init_signs) and 0 <= v < len(init_signs) and u != v:
                    parsed_e.append((min(u, v), max(u, v), w))
            init_edges = parsed_e

    signs_rv = reactive.value(init_signs)
    edges_rv = reactive.value(init_edges)
    pos_rv = reactive.value(circle_positions(len(init_signs)))
    edge_start_rv = reactive.value(None)

    @reactive.effect
    @reactive.event(input.preset)
    def _on_preset():
        k = input.preset()
        if k in PRESETS:
            _, s, e = PRESETS[k]
            signs_rv.set(list(s))
            edges_rv.set(list(e))
            pos_rv.set(circle_positions(len(s)))
            edge_start_rv.set(None)

    @reactive.effect
    @reactive.event(input.negate_btn)
    def _on_negate():
        signs_rv.set([-s for s in signs_rv()])

    @reactive.effect
    @reactive.event(input.add_v_btn)
    def _on_add_v_btn():
        s = list(signs_rv())
        p = dict(pos_rv())
        idx = len(s)
        ang = (idx * 2.39996) % (2 * math.pi)
        p[idx] = (0.85 * math.cos(ang), 0.85 * math.sin(ang))
        s.append(1)
        signs_rv.set(s)
        pos_rv.set(p)

    @reactive.effect
    @reactive.event(input.plot_graph_click)
    def _on_click():
        clk = input.plot_graph_click()
        if not clk or "x" not in clk or "y" not in clk:
            return
        cx, cy = float(clk["x"]), float(clk["y"])
        pos = pos_rv()
        mode = input.edit_mode()
        if mode == "add_v":
            s = list(signs_rv())
            p = dict(pos)
            p[len(s)] = (cx, cy)
            s.append(1)
            signs_rv.set(s)
            pos_rv.set(p)
            return
        nearest, min_d = None, 1e9
        for i, (px, py) in pos.items():
            d = math.hypot(px - cx, py - cy)
            if d < min_d:
                min_d = d
                nearest = i
        if nearest is None or min_d > 0.25:
            edge_start_rv.set(None)
            return
        if mode == "sign":
            s = list(signs_rv())
            s[nearest] = -s[nearest]
            signs_rv.set(s)
        elif mode == "del_v":
            s = list(signs_rv())
            if len(s) <= 2:
                return
            del s[nearest]
            new_e = []
            for u, v, w in edges_rv():
                if u == nearest or v == nearest:
                    continue
                nu = u - 1 if u > nearest else u
                nv = v - 1 if v > nearest else v
                new_e.append((min(nu, nv), max(nu, nv), w))
            new_p = {(k - 1 if k > nearest else k): xy for k, xy in pos.items() if k != nearest}
            signs_rv.set(s)
            edges_rv.set(new_e)
            pos_rv.set(new_p)
            edge_start_rv.set(None)
        elif mode == "edge":
            st = edge_start_rv()
            if st is None:
                edge_start_rv.set(nearest)
            elif st == nearest:
                edge_start_rv.set(None)
            else:
                u, v = min(st, nearest), max(st, nearest)
                e_list = list(edges_rv())
                if any((a, b) == (u, v) for a, b, _ in e_list):
                    e_list = [(a, b, w) for a, b, w in e_list if (a, b) != (u, v)]
                else:
                    w_raw = input.edge_weight().strip().lower()
                    if w_raw in ("oo", "inf", "∞"):
                        w = math.inf
                    else:
                        try:
                            w = max(2.0, float(Fraction(w_raw)))
                        except ValueError:
                            try:
                                w = max(2.0, float(w_raw))
                            except ValueError:
                                w = 3.0
                    e_list.append((u, v, w))
                edges_rv.set(e_list)
                edge_start_rv.set(None)

    @render.ui
    def mode_hint_ui():
        m = input.edit_mode()
        st = edge_start_rv()
        if m == "sign":
            msg = "Click any vertex to flip its sign (+ Garnet ↔ − Lean Blue)."
        elif m == "edge":
            msg = f"Selected vertex {st}; click a second vertex to toggle edge." if st is not None else "Click two vertices to add or remove an edge."
        elif m == "add_v":
            msg = "Click empty canvas space to add a (+) vertex."
        else:
            msg = "Click any vertex to delete it."
        return ui.p(msg, class_="text-muted small mb-0 mt-1")

    @render.plot
    def plot_graph():
        s = signs_rv()
        e = edges_rv()
        pos = pos_rv()
        st = edge_start_rv()
        fig, ax = plt.subplots(figsize=(7.5, 4.0))
        for u, v, w in e:
            x0, y0 = pos[u]
            x1, y1 = pos[v]
            ax.plot([x0, x1], [y0, y1], color="#333333", lw=2.3, zorder=1)
            if w != 3:
                ax.text((x0 + x1) / 2, (y0 + y1) / 2, "∞" if math.isinf(w) else f"{w:g}", fontsize=9, bbox=dict(fc="white", ec="#cbd5e1", pad=1))
        for i in range(len(s)):
            x, y = pos[i]
            c = GARNET if s[i] > 0 else LEAN_BLUE
            ec = "#f59e0b" if st == i else "#ffffff"
            ax.scatter([x], [y], s=700, color=c, edgecolors=ec, linewidths=3 if st == i else 2, zorder=2)
            ax.text(x, y, f"{i}\n({'+' if s[i] > 0 else '−'})", color="white", ha="center", va="center", fontsize=8.5, fontweight="bold", zorder=3)
        xs = [p[0] for p in pos.values()]
        ys = [p[1] for p in pos.values()]
        ax.scatter([], [], s=120, color=GARNET, label="(+) Positive")
        ax.scatter([], [], s=120, color=LEAN_BLUE, label="(−) Negative")
        ax.legend(loc="upper center", bbox_to_anchor=(0.5, 1.06), ncol=2, frameon=False, fontsize=9.5)
        ax.set_xlim(min(xs) - 0.35, max(xs) + 0.35)
        ax.set_ylim(min(ys) - 0.35, max(ys) + 0.35)
        ax.set_aspect("equal")
        ax.axis("off")
        return fig

    @render.plot
    def plot_spectrum():
        s, e = signs_rv(), edges_rv()
        B = build_B(s, e)
        bp = bipartition(len(s), e)
        order = (bp[0] + bp[1]) if (input.order_mode() == "bipartite" and bp is not None) else list(range(len(s)))
        _, w = principal_element(B, order)
        eigs = np.linalg.eigvals(w)
        fig, ax = plt.subplots(figsize=(5.0, 3.4))
        theta = np.linspace(0, 2 * np.pi, 256)
        ax.plot(np.cos(theta), np.sin(theta), "--", color="#94a3b8", lw=1.2)
        ax.axhline(0, color="#cbd5e1", lw=0.8)
        ax.axvline(0, color="#cbd5e1", lw=0.8)
        ax.scatter(eigs.real, eigs.imag, s=75, color=GARNET, zorder=3)
        lim = max(1.3, float(np.max(np.abs(eigs))) * 1.15)
        ax.set_xlim(-lim, lim)
        ax.set_ylim(-lim, lim)
        ax.set_aspect("equal")
        ax.set_title(f"Spectrum (ρ = {float(np.max(np.abs(eigs))):.4f})")
        return fig

    @render.ui
    def summary_ui():
        s, e = signs_rv(), edges_rv()
        n = len(s)
        B = build_B(s, e)
        bp = bipartition(n, e)
        _, w_lin = principal_element(B, list(range(n)))
        rho_lin = float(np.max(np.abs(np.linalg.eigvals(w_lin))))
        if bp is not None:
            _, w_bip = principal_element(B, bp[0] + bp[1])
            rho_bip = float(np.max(np.abs(np.linalg.eigvals(w_bip))))
            pct = ((rho_bip / rho_lin) - 1.0) * 100.0 if rho_lin > 0 else 0.0
            bip_html = f"<div class='mb-2'><strong>✓ Bipartite</strong> | <code>ρ_lin={rho_lin:.4f}</code> | <code>ρ_bip={rho_bip:.4f}</code> ({pct:+.1f}%)</div>"
            thm_html = "<li>✅ <strong>Bipartite Principal Element Theorem</strong> applicable (<a href='https://jkylearmstrong.github.io/mixed.sign.coxeter.systems/lean-verification.html' target='_blank'>Learn More →</a>)</li>"
        else:
            bip_html = f"<div class='mb-2'><strong>✗ Not bipartite</strong> | <code>ρ_lin={rho_lin:.4f}</code></div>"
            thm_html = ""
        evals_B = np.linalg.eigvalsh(B)
        p, q = int(np.sum(evals_B > 1e-9)), int(np.sum(evals_B < -1e-9))
        return ui.HTML(
            bip_html
            + f"<div class='small text-muted mb-2'>Vertices: <strong>{n}</strong> · Edges: <strong>{len(e)}</strong> · Signature: <strong>({p}, {q}, {n-p-q})</strong></div>"
            + f"<ul class='small mb-0 ps-3'><li>✅ <strong>Theorem 1.1 (Howlett's Formula)</strong> holds (<a href='https://jkylearmstrong.github.io/mixed.sign.coxeter.systems/lean-verification.html' target='_blank'>Learn More →</a>)</li>{thm_html}</ul>"
        )

    @render.ui
    def share_ui():
        s, e = signs_rv(), edges_rv()
        q = urlencode({"signs": "".join("+" if x > 0 else "-" for x in s), "edges": ",".join(f"{u}-{v}:{'oo' if math.isinf(w) else f'{w:g}'}" for u, v, w in sorted(e))})
        full_url = f"https://jkylearmstrong.github.io/mixed.sign.coxeter.systems/explorer.html?{q}"
        return ui.HTML(
            f"<div class='small'><strong>Shareable Graph Link:</strong><br/>"
            f"<input type='text' class='form-control form-control-sm mt-1 mb-1' readonly value='{full_url}' onclick='this.select();' />"
            f"<a href='{full_url}' target='_blank'>Open shared link ↗</a></div>"
        )

    @render.text
    def eigs_out():
        s, e = signs_rv(), edges_rv()
        B = build_B(s, e)
        bp = bipartition(len(s), e)
        order = (bp[0] + bp[1]) if (input.order_mode() == "bipartite" and bp is not None) else list(range(len(s)))
        _, w = principal_element(B, order)
        eigs = sorted(np.linalg.eigvals(w), key=lambda z: -abs(z))
        lines = [f"Active order: {order}", "Eigenvalues of w = -U^(-1) U^T:"]
        lines.extend(f"  {z.real:+.6f} {z.imag:+.6f}j  (|λ| = {abs(z):.6f})" for z in eigs)
        return "\n".join(lines)


app = App(app_ui, server)
